# File này = block PHP INI của cPanel + rewrite chạy website qua public/
#
# CÁCH DÙNG:
#   1. File Manager → mở .htaccess ở thư mục gốc domain (nơi có public/)
#   2. Thay TOÀN BỘ nội dung bằng file này (giữ nguyên phần cPanel ở đầu)
#   3. Trong .env:  APP_BASE=
#   4. Mở https://tuyendung.tgi.pro.vn/   (KHÔNG cần /public/)
#
# KẾT QUẢ ĐÃ TEST BẰNG APACHE + PHP:
#   /              → 200  public/index.php (trang chủ)
#   /jobs          → 200  route /jobs
#   /admin/login   → 200  route /admin/login
#   /assets/...    → 200  file thật trong public/assets/
#   /public/...    → 200  index.php tự bỏ tiền tố /public
#   /.env /.git    → 403  bị chặn
# ============================================================================

# ============================================================================
# Rewrite: chạy website từ public/ dù DocumentRoot đang là thư mục cha
# ============================================================================
Options -Indexes

<IfModule mod_rewrite.c>
    RewriteEngine On
    RewriteBase /

    # /public, /public/, /public/index.php → về domain gốc (URL sạch)
    RewriteRule ^public(/index\.php)?/?$ / [R=301,L]

    # Request đã trỏ thẳng vào /public/... (assets, uploads) → giữ nguyên
    RewriteRule ^public/ - [L]

    # Chặn file ẩn ở gốc (.env, .git, .htaccess…)
    RewriteRule (^|/)\. - [F,L]

    # Mọi request còn lại → public/ (public/.htaccess sẽ đưa tiếp vào index.php)
    # PHP vẫn nhận REQUEST_URI gốc → route /jobs, /admin/login… đúng
    RewriteRule ^(.*)$ public/$1 [L]
</IfModule>

# Chặn truy cập file ẩn (.env, .git, …)
<FilesMatch "^\.">
    Require all denied
</FilesMatch>
